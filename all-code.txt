let getDisplay = document.getElementById("display");
let getSubmit = document.getElementById("submit");
let getCity = document.getElementById("city");

async function getWeather() {
  const gotCity = getCity.value.trim();
  if (!gotCity) {
    getDisplay.textContent = "Please Enter a city name";
    return;
  }

  getDisplay.textContent = "Loading...";
  getSubmit.disabled = true;

  try {
    const geoRes = await fetch(
      `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(gotCity)}&count=1&language=en&format=json`,
    );
    if (!geoRes.ok) {
      throw new Error("Geo coding failed");
    }
    const geoData = await geoRes.json();
    if (!geoData.results || geoData.results.length === 0) {
      getDisplay.textContent = "City not Found";
      return;
    }
    const matchedPlace = geoData.results[0];
    const foundName = matchedPlace.name.toLowerCase();
    const { latitude, longitude } = matchedPlace;

    const weatherRes = await fetch(
      `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current=temperature_2m`,
    );

    const weatherData = await weatherRes.json();
    getDisplay.textContent = `${foundName},${weatherData.current.temperature_2m}Â°C`;
  } catch (error) {
    getDisplay.textContent = "Unable to fetch weather. Try again.";
  } finally {
    getSubmit.disabled = false;
  }
}

getSubmit.addEventListener("click", getWeather);

getCity.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    getWeather();
  }
});

let getDisplay = document.getElementById("display");
let getfrom = document.getElementById("from");
let getTo = document.getElementById("to");
let getsubmit = document.getElementById("submit");
let getNumber = document.getElementById("number");
let myform = document.getElementById("myform");

const apiKey = "9b8fb611ba16c087d7844b47";

function syncDropdowns(changedSelect, targetSelect) {
  const selectedValue = changedSelect.value;
  for (const option of targetSelect.options) {
    option.disabled = option.value !== "" && option.value === selectedValue;
  }
}

getfrom.addEventListener("change", () => {
  syncDropdowns(getfrom, getTo);
});

getTo.addEventListener("change", () => {
  syncDropdowns(getTo, getfrom);
});

async function calc(event) {
  event.preventDefault();
  const gotfrom = getfrom.value;
  const gotTo = getTo.value;
  const gotNumber = Number(getNumber.value);

  if (!gotfrom || !gotTo || gotNumber <= 0) {
    getDisplay.textContent = "Enter valid amount";
    return;
  }

  getDisplay.textContent = "Converting...";
  getsubmit.disabled = true;

  try {
    const response = await fetch(
      `https://v6.exchangerate-api.com/v6/${apiKey}/pair/${gotfrom}/${gotTo}/${gotNumber}`,
    );

    if (!response.ok) {
      throw new Error("Conversion request failed");
    }
    const data = await response.json();
    if (data.result !== "success") {
      throw new Error("API error");
    }
    const result = Number(data.conversion_result).toFixed(2);

    getDisplay.textContent = `${result} ${gotTo}`;
  } catch (error) {
    getDisplay.textContent = "Error fetching rate";
  } finally {
    getsubmit.disabled = false;
  }
}

myform.addEventListener("submit", calc);

const quizCard = document.querySelector(".quiz");
const form = document.getElementById("form");
const getNext = document.getElementById("next");
const getDisplay = document.getElementById("display");
const getReset = document.getElementById("reset");
const getError = document.getElementById("error");
const getProgress = document.getElementById("progress");

let count = 1;
let right = 0;
let isSubmited = false;

const totalQuestions = 5;
const rightAnswers = [null, "1", "2", "4", "3", "2"];

form.addEventListener("change", () => {
  getError.textContent = "";
});

function result(event) {
  event.preventDefault();

  if (!isSubmited) {
    const checked = document.querySelector(`input[name="qes${count}"]:checked`);
    if (!checked) {
      getError.textContent = "Please select an option first.";
      return;
    }
    getError.textContent = "";
    isSubmited = true;
    if (checked.value === rightAnswers[count]) {
      right++;
      quizCard.classList.add("correct");
    } else {
      quizCard.classList.add("wrong");
    }

    document.querySelectorAll(`input[name="qes${count}"]`).forEach((radio) => {
      radio.disabled = true;
    });

    getNext.textContent = count === totalQuestions ? "Finish" : "Next";
    return;
  }

  quizCard.classList.remove("correct", "wrong");
  isSubmited = false;

  document.getElementById("set" + count).classList.add("none");
  document.getElementById("set" + (count + 1)).classList.remove("none");

  count++;
  if (count <= totalQuestions) {
    getProgress.textContent = `${count} / ${totalQuestions}`;
    getNext.textContent = "Submit";
  } else {
    getProgress.classList.add("none");
    getNext.classList.add("none");
    getDisplay.textContent = `Right answers:${right} / ${totalQuestions}`;
  }
}

function resetBack() {
  document.getElementById(`set${count}`).classList.add("none");
  count = 1;
  right = 0;
  isSubmited = false;

  quizCard.classList.remove("correct", "wrong");

  form.querySelectorAll("input[type='radio']").forEach((radio) => {
    radio.disabled = false;
  });

  document.getElementById("set1").classList.remove("none");

  getProgress.textContent = `1 / ${totalQuestions}`;
  getProgress.classList.remove("none");
  getError.textContent = "";
  getDisplay.textContent = "";
  getNext.textContent = "Submit";
  getNext.classList.remove("none");
}

form.addEventListener("submit", result);
getReset.addEventListener("click", resetBack);

  let expVal = 0;

  getHistory.querySelectorAll(".t-sets").forEach((el) => {
    el.remove();
  });

  entries.forEach((entry) => {
    const isIncome = entry.type === "income";
    if (isIncome) {
      incVal += entry.amount;
    } else {
      expVal += entry.amount;
    }

    const clone = template.content.cloneNode(true);
    const amountEl = clone.querySelector(".t-show-number");
    clone.querySelector(".t-setname").textContent = entry.desc;
    clone.querySelector(".t-choosen").textContent = entry.type;
    amountEl.textContent = `${isIncome ? "+" : "-"} â‚¹ ${entry.amount.toLocaleString("en-IN")}`;
    amountEl.classList.add(isIncome ? "green" : "red");

    clone.querySelector(".delete").addEventListener("click", () => {
      entries = entries.filter((item) => item.id !== entry.id);
      saveAndRender();

    getHistory.prepend(clone);

const getDisplay = document.getElementById("display");
const container = document.querySelector("#calc");

const operators = ["+", "-", "*", "/"];

container.addEventListener("click", (event) => {
  if (event.target.tagName !== "BUTTON") {
    return;
  }

  const btn = event.target;

  if (btn.id === "clear") {
    getDisplay.textContent = "";
  }

  if (btn.id === "result") {
    const expression = getDisplay.textContent.trim();
    const lastChar = expression.slice(-1);
    if (!expression || operators.includes(lastChar)) {
      return;
    }

    try {
      const result = new Function(`return ${expression}`)();
      getDisplay.textContent = Number.isFinite(result) ? result : "Error";
    } catch {
      getDisplay.textContent = "Error";
    }

    return;
  }

  const value = btn.value;
  if (!value) {
    return;
  }

  const currentText = getDisplay.textContent;
  const lastChar = currentText.slice(-1);

  if (currentText === "Error") {
    getDisplay.textContent = value;
    return;
  }

  if (operators.includes(lastChar) && operators.includes(value)) {
    getDisplay.textContent = currentText.slice(0, -1) + value;
  }

  getDisplay.textContent = getDisplay.textContent + value;
});

function getLowerCases() {
  const combo = "abcdefghijklmnopqrstuvwxyz";

  const combo = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";

  const combo = "0123456789";

    resultChars.push(allFunctions[i]());
  }

  for (let i = resultChars.length; i < finalLength; i++) {
    const randomIndex = Math.floor(Math.random() * allFunctions.length);
    resultChars.push(allFunctions[randomIndex]());
  }

  for (let i = resultChars.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [resultChars[i], resultChars[j]] = [resultChars[j], resultChars[i]];
  }

  getDisplay.textContent = resultChars.join("");

function pad(num) {

function stopWatch() {
  ms++;
  if (ms === 100) {
    ms = 0;
    sec++;
  }

  if (sec === 60) {
    sec = 0;
    min++;
  }

  if (min === 60) {
    min = 0;
    hr++;
  }

  getDisplay.textContent = `${pad(hr)} : ${pad(min)} : ${pad(sec)} : ${pad(ms)}`;
}

      tHr--;

      tSec = 59;

      tMin--;
      tSec = 59;

    tSec--;

            this.builder.pushUInt(this.value, bitWidth);

        return isInline(this.type) ? Math.max(width, this.width) : this.width;

        let existing_vtable = 0;

            const vt2 = this.bb.capacity() - this.vtables[i];

                for (let j = SIZEOF_SHORT; j < len; j += SIZEOF_SHORT) {
